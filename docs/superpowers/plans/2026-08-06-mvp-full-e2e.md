# LineageGuard MVP — Full E2E Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver a working MVP that runs the canonical rename scenario end-to-end (PR diff → DataHub context → BLOCK → migration → validation → GitHub PR) with a Mission Control UI.

**Architecture:** Worker-driven pipeline orchestration. The agent package contains pure step functions; the worker claims runs from Postgres via lease and drives them through the FSM. Next.js UI subscribes via SSE for live updates. LLM calls go through Vercel AI SDK to OmniRoute (localhost:20128).

**Tech Stack:** TypeScript/Node 24, Vitest, Next.js 16, Tailwind 4, shadcn/ui, Vercel AI SDK, PostgreSQL, DataHub MCP (stdio), Python 3.12 (seeding only)

## Global Constraints

- Node.js 24 LTS, TypeScript 5.9 strict mode
- Zod for all runtime validation (match existing domain patterns)
- Vitest for unit/integration tests
- Biome for format/lint (already configured)
- All new packages use ESM (`"type": "module"`)
- ID formats: `run_[a-f0-9]{24}`, `chg_[a-f0-9]{24}`, `evt_[a-f0-9]{24}`
- Commits: conventional commits, co-authored with Claude
- No external API calls without explicit env var configuration
- Match existing code style: strict zod schemas, `.js` extension in imports

```typescript
    path: z.string().min(1).max(240),
    content: z.string().min(1).max(100_000),
    operation: z.enum(["CREATE", "MODIFY"]),
  })).min(1).max(20),
  summary: z.string().min(1).max(1000),
});
```

```typescript
import type { ImpactContext } from "@lineageguard/domain";

export function migrationPlanPrompt(input: {
  table: string;
  field: string;
  operation: string;
  newName: string;
  consumers: Array<{ name: string; type: string; criticality: string }>;
}): string {
  const consumerList = input.consumers
    .map((c) => `- ${c.name} (${c.type}, ${c.criticality})`)
    .join("\n");

  return `You are a database migration expert. Generate an expand-migrate-contract migration plan.

## Proposed Change
Table: ${input.table}
Operation: ${input.operation} column "${input.field}" to "${input.newName}"

## Downstream Consumers Found
${consumerList}

## Requirements
1. The migration must be backward-compatible
2. Old column must remain accessible during transition
3. All known consumers must be updated
4. Add assertions to verify data integrity
5. Include a deprecation timeline
6. Include rollback instructions

Return a JSON object matching the migration plan schema.`;
}

export function migrationPatchPrompt(input: {
  plan: { steps: Array<{ action: string; description: string; targetPath?: string }> };
  table: string;
  field: string;
  newName: string;
  existingModels: string[];
}): string {
  const steps = input.plan.steps
    .map((s, i) => `${i + 1}. [${s.action}] ${s.description}`)
    .join("\n");

  return `You are a database migration code generator. Generate the actual migration artifacts.

## Migration Plan
${steps}

## Context
- Table: ${input.table}
- Old column: ${input.field}
- New column: ${input.newName}
- Existing dbt models to update: ${input.existingModels.join(", ") || "none"}

## Output Requirements
Generate actual SQL and dbt code for each artifact. Be precise and production-ready.
- SQL migrations go in walkthrough/migrations/
- dbt models go in walkthrough/models/
- dbt tests go in walkthrough/tests/
- Documentation goes in docs/migrations/

Return a JSON object matching the migration patch schema.`;
}
```

```typescript
    prompt: migrationPlanPrompt({
      table: input.table,
      field: input.field,
      operation: "RENAME",
      newName: input.newName,

    }),
  });

  return { plan };
```

Create `packages/agent/src/pipeline.ts`:

```typescript
import type { LanguageModelV1 } from "ai";
import type { DataHubContextPort } from "@lineageguard/datahub";
import type { RunStatus } from "@lineageguard/domain";
import { parseChange, type ParseChangeInput } from "./steps/parse-change.js";
import { baselineAssess } from "./steps/baseline-assess.js";
import { collectContext } from "./steps/collect-context.js";
import { decideRisk } from "./steps/decide-risk.js";
import { planMigration } from "./steps/plan-migration.js";
import { generatePatch } from "./steps/generate-patch.js";
import type { StepContext } from "./steps/index.js";

export interface AgentPipelineConfig {
  datahub: DataHubContextPort;
  llm: LanguageModelV1;
  workerId: string;
  clock: () => Date;
}

export interface RunInput {
  runId: string;
  repository: string;
  baseSha: string;
  headSha: string;
  patch: string;
  table: string;
  field: string;
  newName: string;
}

export interface PipelineResult {
  runId: string;
  finalStatus: RunStatus;
  baselineDecision: string;
  groundedDecision: string;
  consumersFound: number;
  artifactsGenerated: number;
}

export function createAgentPipeline(config: AgentPipelineConfig) {
  const ctx: StepContext = {
    runId: "",
    workerId: config.workerId,
    llm: config.llm,
    datahub: config.datahub,
    clock: config.clock,
  };

  return {
    async execute(input: RunInput): Promise<PipelineResult> {
      ctx.runId = input.runId;

      // Step 1: Parse change
      const { change } = await parseChange(ctx, {
        repository: input.repository,
        baseSha: input.baseSha,
        headSha: input.headSha,
        patch: input.patch,
      });

      // Step 2: Baseline assessment
      const changeId = change.changeId ?? `chg_${"0".repeat(24)}`;
      const { baseline } = await baselineAssess(ctx, changeId);

      // Step 3: Collect DataHub context
      const { context } = await collectContext(ctx, changeId);

      // Step 4: Decide risk
      const { comparison } = await decideRisk(ctx, {
        changeId,
        baseline,
        context,
      });

      // Step 5: Plan migration (only if BLOCK or REVIEW)
      let artifactsGenerated = 0;
      if (comparison.grounded.decision !== "ALLOW") {
        const { plan } = await planMigration(ctx, {
          context,
          table: input.table,
          field: input.field,
          newName: input.newName,
        });

        // Step 6: Generate patch
        const { candidate } = await generatePatch(ctx, {
          plan,
          table: input.table,
          field: input.field,
          newName: input.newName,
        });
        artifactsGenerated = candidate.artifacts?.length ?? 0;
      }

      return {
        runId: input.runId,
        finalStatus: comparison.grounded.decision === "ALLOW" ? "COMPLETED" : "VALIDATED",
        baselineDecision: baseline.decision,
```

```tsx
export interface RunEvent {
  runId: string;
  status: string;
  timestamp: string;
  detail?: string;
}

export function useRunEvents(runId: string) {
  const [events, setEvents] = useState<RunEvent[]>([]);
  const [currentStatus, setCurrentStatus] = useState<string>("CREATED");

  useEffect(() => {
    const source = new EventSource(`/api/runs/${runId}/events`);
    source.onmessage = (e) => {
      const event: RunEvent = JSON.parse(e.data);
      setEvents((prev) => [...prev, event]);
      setCurrentStatus(event.status);
    };
    return () => source.close();
  }, [runId]);

  return { events, currentStatus };
}
```

```tsx
  { status: "VALIDATED", label: "Validated" },
  { status: "REVIEW_ARTIFACT_CREATED", label: "PR Created" },
  { status: "COMPLETED", label: "Complete" },
];

export function RunTimeline({ currentStatus }: { currentStatus: string }) {
  const currentIdx = steps.findIndex((s) => s.status === currentStatus);

  return (
    <div className="flex items-center gap-1 overflow-x-auto py-4 px-6">
      {steps.map((step, i) => {
        const completed = i < currentIdx;
        const active = i === currentIdx;
        return (
          <div key={step.status} className="flex items-center">
            <motion.div
              className={`flex flex-col items-center ${active ? "scale-110" : ""}`}
              animate={{ scale: active ? 1.05 : 1 }}
            >
              <div
                className={`w-3 h-3 rounded-full ${
                  completed
                    ? "bg-status-pass"
                    : active
                      ? "bg-status-info ring-2 ring-status-info/30"
                      : "bg-muted"
                }`}
              />
              <span className={`text-[10px] mt-1 whitespace-nowrap ${active ? "text-foreground font-medium" : "text-muted-foreground"}`}>

            </motion.div>

          </div>
```

```typescript
import { createAgentModel, agentLLMConfigFromEnv, createAgentPipeline } from "@lineageguard/agent";
import { createOfficialLiveDataHubContextPort } from "@lineageguard/datahub";
import type { PipelineResult } from "@lineageguard/agent";

async function main() {
  console.log("=== LineageGuard MVP Demo ===\n");

  // 1. Configure
  const llmConfig = agentLLMConfigFromEnv();
  const llm = createAgentModel(llmConfig);
  const datahubGmsUrl = process.env.DATAHUB_GMS_URL ?? "http://localhost:8080";
  const datahubToken = process.env.DATAHUB_TOKEN ?? "";

  console.log(`DataHub: ${datahubGmsUrl}`);
  console.log(`LLM: ${llmConfig.baseURL} (model: ${llmConfig.model})\n`);

  // 2. Create pipeline
  const pipeline = createAgentPipeline({
    datahub: { collect: async (input) => {
      console.log("  [DataHub] Collecting context for", input.changeId);
      const port = createOfficialLiveDataHubContextPort({
        gmsUrl: datahubGmsUrl,
        token: datahubToken,
      });
      return port.collect(input);
    }} as any,
    llm,
    workerId: "demo",
    clock: () => new Date(),
  });

  // 3. Execute canonical scenario
  console.log("--- Running canonical scenario: RENAME customer_id → buyer_id ---\n");

  const result: PipelineResult = await pipeline.execute({
    runId: "run_000000000000000000000001",
    repository: "greatkich/lineageguard",
    baseSha: "a".repeat(40),
    headSha: "b".repeat(40),
    patch: "ALTER TABLE commerce.orders RENAME COLUMN customer_id TO buyer_id;",
    table: "commerce.orders",
    field: "customer_id",
    newName: "buyer_id",
  });

  // 4. Report
  console.log("\n=== Results ===");
  console.log(`Baseline decision:   ${result.baselineDecision}`);
  console.log(`Grounded decision:   ${result.groundedDecision}`);
  console.log(`Decision changed:    ${result.baselineDecision !== result.groundedDecision ? "YES" : "NO"}`);
  console.log(`Consumers found:     ${result.consumersFound}`);
  console.log(`Artifacts generated: ${result.artifactsGenerated}`);
  console.log(`Final status:        ${result.finalStatus}`);
  console.log("\n=== Demo Complete ===");

  if (result.groundedDecision === "BLOCK" && result.consumersFound >= 4) {
    console.log("\n✓ SUCCESS: DataHub changed the decision from ALLOW to BLOCK");
    process.exit(0);
  } else {
    console.error("\n✗ UNEXPECTED: Decision did not change as expected");
    process.exit(1);
  }
}

main().catch((err) => {
  console.error("Demo failed:", err);
  process.exit(1);
});
```
